

ALTER Procedure [dbo].[sp_SubstationDataWithId]    
(    
--@ID int  
@ID  nvarchar(50)
)    
As    
Begin    
Select Distinct
CD.Email As ContractorEmail, CD.ContactNo AS ContractorContactNo,
CASE
    WHEN Wi.AppliedBy = 'SiteOwner' THEN 'SiteOwner'
    ELSE
        CASE
            WHEN CD.LicenceNew IS NOT NULL AND CD.LicenceNew != '' THEN CD.LicenceNew
            ELSE CD.LicenceOld
        END + '/' + CD.Name
END AS ContractorWhoCreated,

CASE
    WHEN Wi.AppliedBy = 'SiteOwner' THEN 'SiteOwner'
    ELSE
        CASE
            WHEN WS.CertificateNew IS NOT NULL AND WS.CertificateNew != '' THEN WS.CertificateNew
            ELSE WS.CertificateOld
        END + '/' + WS.Name
END AS SupervisorWhoCreated,
CASE  
WHEN WI.PremisesType = 'Other'
THEN WI.OtherPremises
ELSE  WI.PremisesType END AS Permises,
CASE WHEN ST.LightningArrestorLocation = 'other' THEN ST.OtherLALocation
ELSE  ST.LightningArrestorLocation
END AS LightningArrestorLocationOther ,
ST.TestReportId+' / '+ WI.Id as SubStationId,
 ST.TestReportId as TestReportId,
 CONVERT(VARCHAR(10), ST.ApprovedDate, 105) as ApprovalDate,
 CONVERT(VARCHAR(10), WI.WorkStartDate, 105) as WorkStartDate,
CONVERT(VARCHAR(10), WI.CompletionDate, 105) as CompletionDate,
CONVERT(VARCHAR(10), WI.CreatedDate, 105) AS SubmittedDate,
 WI.Id as WorkIntimationId,

 WI.ApplicantType,
WI.District,
WI.Inspectiontype,

WI.NameOfOwner,
WI.NameOfAgency, 
WI.ContactNo,
WI.Address,
CA.Area,
ST.TransformerSerialNumber,
ST.TransformerCapacityType,
ST.TranformerType,
ST.YearOfManufacturing,
ST.TypeofHTPrimarySideSwitch,
ST.TransformerCapacity,
ST.TranformerType,
ST.Count,
ST.PrimaryVoltage,
ST.SecondoryVoltage,
ST.OilCapacity,
ST.BreakDownVoltageofOil,
ST.HtInsulationHVEarth,
ST.LtInsulationLVEarth,
ST.LowestvaluebetweenHTLTSide,
ST.NumberOfEarthing,
ST.EarthingType1,
ST.Valueinohms1,
ST.UsedFor1,
ST.OtherEarthing1,
ST.EarthingType2,
ST.Valueinohms2,
ST.UsedFor2,

ST.UsedFor10,

ST.EarthingType11,
ST.Valueinohms11,
ST.UsedFor11,
ST.OtherEarthing11,
ST.EarthingType12,

From tbl_SubstationTransformer As ST  
iNNER jOIN tbl_WorkIntimation as WI On ST.IntimationId = WI.id  
Left Join tbl_ContractorDetails AS CD ON CD.UserId = WI.CreatedBy  
LEFT JOIN tbl_CEIAreaCovered AS CA ON WI.District = CA.AreaCovered
Left join tbl_WiremanandSuperviserDetails AS WS on WS.UserId = ST.CreatedBy
Left join tbl_VoltageLevel AS VL On WI.VoltageLevel=VL.Voltagelevel
Where 
--ST.ID = @ID    
ST.TestReportId=@ID
End
