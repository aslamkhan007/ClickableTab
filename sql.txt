
ALTER PROCEDURE [dbo].[sp_InsertSwitchingStationData_Existing_HavingPreviousReport]
(
    @Count INT,
    @IntimationId NVARCHAR(50),
    @StationVoltage NVARCHAR(20),
    @SwitchingStationName NVARCHAR(50),
    @BreakerType NVARCHAR(50),
    @OtherbreakerType NVARCHAR(50) = NULL,
    @TotalBreakers NVARCHAR(50) ,
    @StationTransformerCapacity INT,
    @SerialNo NVARCHAR(30),
    @LastInspectionDate DATE,
    @ApplicantType NVARCHAR(50),	
    @VoltageLevel NVARCHAR(50),
    @District NVARCHAR(50),
    @Division NVARCHAR(50),
    @Inspectiontype NVARCHAR(50),
    @CreatedBy NVARCHAR(50),
    @ReturnStatus INT OUTPUT
)
AS
BEGIN TRY
    BEGIN TRANSACTION;

    DECLARE @TReportID NVARCHAR(50);
SET @TReportID = 'TSS-' + RIGHT('00000' + CAST(ISNULL((SELECT MAX(CONVERT(BIGINT, ID)) FROM tbl_SwitchingStation), 0) + 1 AS NVARCHAR(50)), 5);
print @TReportID

    INSERT INTO tbl_SwitchingStation
    ( 
        TestReportId,
        Count,
        IntimationId,
        SerialNoofSwitchingStation,
        VoltageLevelofSwitchingStation,
        NamePlaceofSwitchingStation,
        TypeofBreaker,
        OtherBreakerType,
        TotalNoofBreakers,
        CapacityofStationTransformerInKva,
        ApprovedDate,
        Status,
        CreatedBy,
        CreatedDate,
		LastInspectionDate
    )
    VALUES  
    ( 
        @TReportID,
        @Count,
        @IntimationId,
        @SerialNo,
        @StationVoltage,
        @SwitchingStationName,
        @BreakerType,
        @OtherbreakerType,
        @TotalBreakers,
        @StationTransformerCapacity,
        GETDATE(),  
        1, 
        @CreatedBy,
        GETDATE(),
		@LastInspectionDate
    );  

    INSERT INTO tbl_Inspection
    (TestRportId, IntimationId, TestReportCount, ApplicantType, InstallationType, VoltageLevel,
     District, Division, ApplicationStatus, Inspectiontype, CreatedBy, CreatedDate, ApprovedDate, 
     InspectionNewOrExist, PreviousInspectionDate, SystemGenerated)
    VALUES 
    (@TReportID, 
     @IntimationId, 
     @Count, 
     @ApplicantType, 
     'Switching Station',
     @VoltageLevel,
     @District, 
     @Division, 
     'Approved',
     @Inspectiontype, 
     @CreatedBy,	
     GETDATE(),
     @LastInspectionDate,
     'Existing',  
     'Yes', 
     'No');

    UPDATE tbl_InstallationHistory
    SET Inspection = 'Generated',
        InspectionCreatedBy = @CreatedBy,
        TestReportApprovedOrNot = 'Submit'
    WHERE IntimationId = @IntimationId
          AND Count = @Count
          AND Typeofinstallation = 'Switching Station';

    UPDATE tbl_SwitchingStation  
    SET ApprovedOrRejectFromContractor = 'Submit',  
        Approveddate = GETDATE()  
    WHERE IntimationId = @IntimationId  
          AND [Count] = @Count
          AND Status = 1;

    -- Determine return status based on inspection history
    IF EXISTS (SELECT 1 FROM tbl_InstallationHistory 
               WHERE IntimationId = @IntimationId
               AND TestReportCreatedby = 'Pending')
    BEGIN 
        SET @ReturnStatus = 1;
    END 
    ELSE 
    BEGIN 
        SET @ReturnStatus = 0;
    END 

    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    ROLLBACK TRANSACTION;

END CATCH;



go



ALTER PROCEDURE [dbo].[sp_InsertSubstationTransformerData_Existing_HavingPreviousReport]
    (
      @IdUpdate INT = NULL ,
      @Count INT ,
      @IntimationId NVARCHAR(50) ,
      @TransformerSerialNumber NVARCHAR(50) ,
      @TransformerCapacityType NVARCHAR(10) ,
      @TransformerCapacity NVARCHAR(50) ,
      @TranformerType NVARCHAR(50) ,
      @YearOfManufacturing NVARCHAR(10) ,
      @PrimaryVoltage NVARCHAR(50) ,
      @SecondoryVoltage NVARCHAR(50) ,
      @MakeType NVARCHAR(100) ,
      @LastInspectionDate DATE ,
      @ApplicantType NVARCHAR(50) ,
      @VoltageLevel NVARCHAR(50) ,
      @District NVARCHAR(50) ,
      @Division NVARCHAR(50) ,
      @Inspectiontype NVARCHAR(50) ,
      @CreatedBy NVARCHAR(10) ,
@NamePlateDocument NVARCHAR(500) = NULL,
@SitephotographDocument NVARCHAR(500) = NULL,
      @SubstationName NVARCHAR(50) = NULL ,
      @ReturnStatus INT OUTPUT 
    )
AS 
    BEGIN  
		

        DECLARE @CheckInputVoltazeValidity VARCHAR(50) ,
            @ApplicantTypeCodeFromTable VARCHAR(100)

			-- Parse the returned values
        DECLARE @Delimiter CHAR(1) = '|'
        DECLARE @DelimiterPosition INT
        DECLARE @FirstValue NVARCHAR(MAX)
        DECLARE @SecondValue NVARCHAR(MAX)
        DECLARE @StaffAssigned VARCHAR(50) 
        DECLARE @ServiceType VARCHAR(50)

			--Added on 18 Feb 2025 to Pass Dynamic Parameters of Intimation type New or Periodic
        DECLARE @InspectionTypeParam NVARCHAR(MAX)
        SET @InspectionTypeParam = ( SELECT InspectionType
                                     FROM   tbl_WorkIntimation
                                     WHERE  id = @IntimationId
                                            AND Status = 1
                                   )
        IF @InspectionTypeParam = 'Existing'
            OR @InspectionTypeParam = '' 
            BEGIN
                SET @InspectionTypeParam = 'Periodic'
            END


        SET @ApplicantTypeCodeFromTable = ( SELECT  ApplicantTypeCode
                                            FROM    tbl_WorkIntimation
                                            WHERE   id = @IntimationId
                                                    AND Status = 1
                                          )
        SET @CheckInputVoltazeValidity = ( SELECT   dbo.fn_Get_TestReport_ConcernedStaffHierarchy(@IntimationId,
                                                              @Count,
                                                              @ApplicantTypeCodeFromTable,
                                                              @InspectionTypeParam,
                                                              2,
                                                              @PrimaryVoltage,
                                                              @PrimaryVoltage,
                                                             -- @SecondoryVoltage,
                                                              @TransformerCapacity,
                                                              @TransformerCapacity,
                                                              @TransformerCapacityType)
                                         ) 
		
        IF ( @CheckInputVoltazeValidity = '' ) 
            BEGIN 
                RAISERROR ('Voltage Level And Concerned Staff Mapping Not Found.', 16, 1 ) ; 				
                RETURN ; 
            END
        ELSE 
            BEGIN 
                IF ( @CheckInputVoltazeValidity <> '' ) 
                    BEGIN
						-- Find the position of the delimiter
                        SET @DelimiterPosition = CHARINDEX(@Delimiter,
                                                           @CheckInputVoltazeValidity)

						-- Extract the first value before the delimiter
                        SET @FirstValue = SUBSTRING(@CheckInputVoltazeValidity,
                                                    1, @DelimiterPosition - 1)

						-- Extract the second value after the delimiter
                        SET @SecondValue = SUBSTRING(@CheckInputVoltazeValidity,
                                                     @DelimiterPosition + 1,
                                                     LEN(@CheckInputVoltazeValidity)
                                                     - @DelimiterPosition)
                        SET @StaffAssigned = @FirstValue 
                        SET @ServiceType = @SecondValue 
                    END
            END 
 --- Changed on 17 jan 2025 for transaction commit error 
        BEGIN TRY
            BEGIN TRANSACTION	

            DECLARE @TReportID NVARCHAR(50)
            SET @TReportID = 'TS-'
                + ISNULL(CAST(( SELECT  MAX(CONVERT(BIGINT, ID))
                                FROM    tbl_SubstationTransformer
                              ) + 1 AS NVARCHAR(50)), '1')
            IF @IdUpdate IS NOT NULL
                AND @IdUpdate != '' 
                BEGIN  
                    SELECT  1 
                END  
            ELSE 
                BEGIN  	
			
------------------------------------------------------- Deactive Existing Test Report --------------------------------------------------
                    UPDATE  tbl_SubstationTransformer
                    SET     Status = 0
                    WHERE   IntimationId = @IntimationId
                            AND Count = @Count
                            AND Status = 1

                    INSERT  INTO tbl_SubstationTransformer
                            ( [Count] ,
                              TestReportId ,
                              IntimationId ,
                              TransformerSerialNumber ,
                              TransformerCapacityType ,
                              TransformerCapacity ,
                              TranformerType ,
                              YearOfManufacturing ,
                              PrimaryVoltage ,
                              SecondoryVoltage ,
                              CreatedDate ,
                              CreatedBy ,
                              SubstationName ,
                              ApprovedOrRejectFromContractor ,
                              Status ,
                              InspectionCount ,
                              ServiceType ,
                              MakeType ,
                              LastInspectionDate,
 NamePlateDocument,
 SitephotographDocument
                            )
                    VALUES  ( @Count ,
                              @TReportID ,
                              @IntimationId ,
                              @TransformerSerialNumber ,
                              @TransformerCapacityType ,
                              @TransformerCapacity ,
                              @TranformerType ,
                              @YearOfManufacturing ,
                              @PrimaryVoltage ,
                              @SecondoryVoltage ,
                              GETDATE() ,
                              @CreatedBy ,
                              @SubstationName ,
                              'Pending' ,
                              1 ,
                              0 ,
                              @ServiceType ,
                              @MakeType ,
                              @LastInspectionDate,
@NamePlateDocument ,
@SitephotographDocument
                            )  
                END  

            UPDATE  tbl_InstallationHistory
            SET     OldTestReportId = TestReportId ,
                    TestReportCreatedby = @CreatedBy ,
                    TestReportId = @TReportID
            WHERE   IntimationId = @IntimationId
                    AND Count = @Count
                    AND Typeofinstallation = 'Substation Transformer'
         
            INSERT  INTO tbl_Inspection
                    ( TestRportId ,
                      IntimationId ,
                      TestReportCount ,
                      ApplicantType ,
                      InstallationType ,
                      VoltageLevel ,
                      District ,
                      Division ,
                      ApplicationStatus ,
                      Inspectiontype ,
                      CreatedBy ,
                      CreatedDate ,
                      ApprovedDate ,
                      InspectionNewOrExist ,
                      PreviousInspectionDate ,
                      SystemGenerated
                    )
							            --ApproximateYears, InspectionNewOrExist
            VALUES  ( @TReportID ,
                      @IntimationId ,
                      @Count ,
                      @ApplicantType ,
                      'Substation Transformer' ,
                      @VoltageLevel ,
                      @District ,
                      @Division ,
                      'Approved' ,
                      @Inspectiontype ,
                      @CreatedBy ,
                      GETDATE() ,
                      @LastInspectionDate ,
                      'Existing' ,
                      'Yes' ,
                      'No' 
                    )
            DECLARE @NewInspectionId INT
            SET @NewInspectionId = SCOPE_IDENTITY()
            UPDATE  tbl_InstallationHistory
            SET     Inspection = 'Generated' ,
                    InspectionCreatedBy = @CreatedBy,
                    InspectionId = @NewInspectionId
            WHERE   IntimationId = @IntimationId
                    AND Count = @Count
                    AND Typeofinstallation = 'Substation Transformer'


-------------------------------------------------------Test Report Approved-----------------------------------------------------------

            UPDATE  tbl_SubstationTransformer
            SET     ApprovedOrRejectFromContractor = 'Submit' ,
                    Approveddate = GETDATE()
            WHERE   IntimationId = @IntimationId
                    AND [Count] = @Count
                    AND Status = 1

            UPDATE  tbl_InstallationHistory
            SET     TestReportApprovedOrNot = 'Submit'
            WHERE   IntimationId = @IntimationId
                    AND [Count] = @Count
                    AND Typeofinstallation = 'Substation Transformer'

            IF EXISTS ( SELECT  1
                        FROM    tbl_InstallationHistory
                        WHERE   IntimationId = @IntimationId
                                AND TestReportCreatedby = 'Pending' ) 
                BEGIN 
                    SET @ReturnStatus = 1
                END 
            ELSE 
                BEGIN 
                    SET @ReturnStatus = 0
                END 
	
            IF @@ERROR = 0 
                BEGIN
                    COMMIT TRANSACTION
                END
        END TRY

        BEGIN CATCH
            ROLLBACK TRANSACTION
        END CATCH

    END



go
