ALTER procedure [dbo].[sp_GetRenewalLiftDataGridOfAllRecords]
(

As
Begin
SELECT Distinct
       LAT.RegistrationNo,LAT.Make,LAT.LiftSrNo,LAT.TypeOfControl,LAT.Capacity,LAT.[Weight],LAT.Lift_Escelator_Id,

          SELECT MAX(LAT2.CreatedDate)
          FROM tbl_liftApproval_Inspections AS LAT2
          WHERE LAT2.RegistrationNo = LAT.RegistrationNo
      )
ORDER BY LAT.CreatedDate DESC;
End
